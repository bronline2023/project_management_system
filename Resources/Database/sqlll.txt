-- 1. સબસ્ક્રિપ્શન પ્લાન (Subscription Plans)
CREATE TABLE IF NOT EXISTS `subscription_plans` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `plan_name` varchar(100) NOT NULL, -- દા.ત. Basic, Premium
  `monthly_price` decimal(10,2) NOT NULL,
  `yearly_price` decimal(10,2) NOT NULL,
  `features` text, -- JSON format માં કઈ કઈ સુવિધા મળશે
  PRIMARY KEY (`id`)
);

-- 2. પોર્ટલ્સ / ડોમેન્સ (White-label Websites)
CREATE TABLE IF NOT EXISTS `portals` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `domain_name` varchar(255) NOT NULL UNIQUE, -- દા.ત. retailer.com અથવા br.online.com
  `owner_id` int(11) NOT NULL, -- જેણે આ પોર્ટલ ખરીદ્યું છે 
  `plan_id` int(11) NOT NULL,
  `status` enum('active','suspended','expired') DEFAULT 'active',
  `expiry_date` date NOT NULL,
  `folder_path` varchar(255) NOT NULL, -- દા.ત. /uploads/portal_12/ (ડેટા અલગ રાખવા)
  `theme_color` varchar(20) DEFAULT '#1e293b',
  `logo_url` varchar(255) DEFAULT NULL,
  PRIMARY KEY (`id`)
);

-- 3. યુઝર હાયરાર્કી (5-Level User Roles)
CREATE TABLE IF NOT EXISTS `users` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `portal_id` int(11) DEFAULT NULL, -- આ યુઝર કયા પોર્ટલનો છે? (Master માટે NULL)
  `parent_id` int(11) DEFAULT NULL, -- આ યુઝર કોની નીચે છે? (રિટેલર -> મેનેજર -> એડમિન)
  `role` enum('master_admin', 'admin', 'manager', 'district_manager', 'retailer') NOT NULL,
  `name` varchar(100) NOT NULL,
  `email` varchar(100) NOT NULL UNIQUE,
  `password` varchar(255) NOT NULL,
  `wallet_balance` decimal(10,2) DEFAULT '0.00',
  PRIMARY KEY (`id`)
);

-- 4. પેમેન્ટ અને રિચાર્જ હિસ્ટ્રી (Payment Gateway Logs)
CREATE TABLE IF NOT EXISTS `transactions` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `portal_id` int(11) DEFAULT NULL,
  `amount` decimal(10,2) NOT NULL,
  `txn_type` enum('recharge', 'subscription_fee', 'service_deduction') NOT NULL,
  `payment_gateway_id` varchar(100) DEFAULT NULL, -- Razorpay/Stripe Order ID
  `status` enum('success','pending','failed') DEFAULT 'pending',
  `created_at` timestamp DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`)
);
